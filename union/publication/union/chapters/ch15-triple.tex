\section{Three Categories, One Bounded Phase Space}
\label{sec:ch15-three}

The bounded phase space $(\Omega, \mu, \phi_t)$ admits three equivalent
descriptions, each corresponding to a different observational language:

\begin{enumerate}
  \item The \textbf{Oscillatory category} $\Osc$: objects are bounded
    oscillators; morphisms are measure-preserving maps between their phase
    spaces.
    Language: trajectories, frequencies, Fourier modes.
    This is the language of classical mechanics.

  \item The \textbf{Categorical (Epistemological) category} $\Cat$:
    objects are receiver quadruples; morphisms are S-value-preserving
    information channels.
    Language: S-entropy, receivers, observations.
    This is the language of information theory.

  \item The \textbf{Partition category} $\Part$: objects are partition
    hierarchies $(\Partition_n)_{n\geq 1}$; morphisms are refinement maps.
    Language: partition cells, depths, selection rules.
    This is the language of quantum mechanics (cells = eigenstates,
    refinement maps = quantum channels).
\end{enumerate}

\section{The Triple Equivalence}
\label{sec:ch15-equivalence}

\begin{theorem}[Triple Equivalence]
  \label{thm:triple}
  There exist explicit functors
  \begin{equation}
    F_{\Osc\Cat} : \Osc \to \Cat, \qquad
    F_{\Cat\Part} : \Cat \to \Part, \qquad
    F_{\Part\Osc} : \Part \to \Osc,
  \end{equation}
  such that the diagram
  \begin{equation}
    \begin{tikzcd}
      \Osc \arrow[r, "F_{\Osc\Cat}"] \arrow[dr, "F_{\Part\Osc}^{-1}"'] &
      \Cat \arrow[d, "F_{\Cat\Part}"] \\
      & \Part \arrow[ul, "F_{\Part\Osc}"]
    \end{tikzcd}
  \end{equation}
  commutes up to natural isomorphism.
  Therefore $\Osc \cong \Cat \cong \Part$.
\end{theorem}

\begin{proof}[Construction of $F_{\Osc\Cat}$]
  Given a bounded oscillator $({\Omega}, \mu, \phi_t)$, define the
  receiver $R_\Omega = (\Omega, \phi_t, \mathcal{V}(0), \mathrm{dec}_\Omega)$
  where $\mathrm{dec}_\Omega$ maps the current visited set to the Fourier
  coefficients of the trajectory.
  The S-value $\mathbf{S}[R_\Omega, q]$ for the question ``which mode is
  the system in?'' equals $S_\mathrm{osc} = \kB\log|\mathcal{V}|$.
  This defines $F_{\Osc\Cat}$.
\end{proof}

\begin{proof}[Construction of $F_{\Cat\Part}$]
  Given a receiver $R = (\Sigma_R, \Phi_R, K_R, \mathrm{dec}_R)$, the
  partition algebra $\mathcal{A}_\infty(R)$ is the partition algebra of
  $\Sigma_R$ (which satisfies BPSL).
  The categorical observation (Consequence~\ref{cons:categorical}) maps each
  state of $\Sigma_R$ to the cell of $\mathcal{A}_\infty(R)$ it occupies.
  This defines $F_{\Cat\Part}$.
\end{proof}

\begin{proof}[Construction of $F_{\Part\Osc}$]
  Given a partition hierarchy $(\Partition_n)$, the Koopman construction
  associates a Hilbert space $\Hilbert_n = L^2(\Omega, \Partition_n, \mu)$
  to each depth $n$, and a unitary operator $U_t$ whose eigenfrequencies are
  $\{\omega_k\}$.
  The bounded oscillator is $(\Omega, \mu, \phi_t)$ where $\phi_t$ is
  reconstructed from the Koopman eigenvectors.
  This defines $F_{\Part\Osc}$.
\end{proof}

\section{Physical Consequences of Triple Equivalence}
\label{sec:ch15-consequences}

The Triple Equivalence is the formal proof that:
\begin{itemize}
  \item \textbf{Classical mechanics $=$ quantum mechanics}: every trajectory
    of a bounded oscillator ($\Osc$) corresponds to an eigenstate of the
    Koopman operator ($\Part$), and vice versa.
    The two descriptions are not approximate; they are equivalent.

  \item \textbf{Information theory $=$ physics}: S-entropy ($\Cat$) and
    partition entropy ($\Part$) are the same quantity computed in different
    languages.

  \item \textbf{GPU computation $=$ physical measurement}: a GPU fragment
    shader performing partition-depth minimisation ($\Part$) on a bounded
    domain is a physical observation apparatus in the sense of category $\Osc$.
    This is the mathematical foundation of Chapter~\ref{ch:gpu}.
\end{itemize}

\section{Triple Entropy Equivalence}
\label{sec:ch15-entropy}

\begin{consequence}{Triple Entropy Equivalence}{tripleent}
  The oscillatory, categorical, and partition entropies are identical:
  \begin{equation}
    \Sk = \St = \Se = \kB \Depth \ln n,
  \end{equation}
  where $\Depth$ is the partition depth and $n$ is the number of cells at
  that depth.
\end{consequence}

\begin{proof}[Proof sketch]
  $\Sk = \kB\log|\mathcal{V}| = \kB\log(C(n)) = \kB\log(2n^2)$
  (oscillatory entropy from visited cells).\\
  $\St = -\sum_i \mu(A_i)\log\mu(A_i)$ for the uniform partition at depth $n$
  $= \kB\log(C(n))$ (categorical/Shannon entropy).\\
  $\Se = \kB\log C(n)$ (partition entropy by definition of the partition
  measure).\\
  All three equal $\kB\log(2n^2) = \kB(\ln 2 + 2\ln n)$,
  which equals $\kB \Depth \ln n$ for the appropriate depth scaling.
\end{proof}
