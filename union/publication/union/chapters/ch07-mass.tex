\section{The Rest Frequency}
\label{sec:ch7-restfreq}

A bounded oscillator has a fundamental frequency $\omega_0$ that persists even
when the oscillator is at rest with respect to a given observer.

\begin{consequence}{Positive Rest Frequency}{restfreq}
  Every bounded oscillator has a strictly positive minimum frequency
  $\omega_0 > 0$, determined by the fundamental mode of the Sturm--Liouville
  problem on $\Omega$.
\end{consequence}

\begin{proof}[Proof sketch]
  On a compact bounded domain, the Laplacian has a discrete spectrum bounded
  away from zero.
  The ground eigenvalue $\lambda_1 > 0$ (Weyl's law: $\lambda_1 \sim
  |\Omega|^{-2/d}$) gives $\omega_0 = c\sqrt{\lambda_1} > 0$.
\end{proof}

\section{Rest Mass from the Dispersion Relation}
\label{sec:ch7-restmass}

\begin{consequence}{Rest Mass}{restmass}
  The rest mass of a bounded oscillator with rest frequency $\omega_0$ is
  \begin{equation}\label{eq:restmass}
    m_0 = \frac{\hbar\omega_0}{c^2}.
  \end{equation}
\end{consequence}

\begin{proof}
  Setting $k = 0$ (zero spatial momentum) in the dispersion
  relation~\eqref{eq:dispersion}:
  $E_0 = \hbar\omega_0$.
  Einstein's relation $E_0 = m_0 c^2$ gives $m_0 = \hbar\omega_0/c^2$.
  Note that $E_0 = m_0 c^2$ is itself derived from the Lorentz group structure
  (Consequence~\ref{cons:lorentz}), not postulated.
\end{proof}

\section{The Loschmidt Principle}
\label{sec:ch7-loschmidt}

\begin{consequence}{Loschmidt Principle and Entropy Monotonicity}{loschmidt}
  The set of partition cells visited by a trajectory up to time $t$,
  \begin{equation}
    \mathcal{V}(t) = \{A \in \mathcal{A}_\infty : \phi_s(x_0) \in A
    \text{ for some } s \leq t\},
  \end{equation}
  is monotonically increasing: $\mathcal{V}(t) \subseteq \mathcal{V}(t')$
  for $t \leq t'$.
  The entropy of the visited set is therefore non-decreasing:
  \begin{equation}
    S(t) = \kB \log|\mathcal{V}(t)| \geq S(t') \quad \text{for } t \leq t'.
  \end{equation}
\end{consequence}

\begin{proof}
  Visitation is a monotone set operation: once a cell is visited, it remains
  in $\mathcal{V}$.
  This is a set-theoretic fact, not a statistical one.
  The second law of thermodynamics follows as a corollary.
\end{proof}

\begin{remark}[Resolution of the Loschmidt Paradox]
  The Loschmidt paradox asks: why does entropy increase if microscopic
  dynamics is time-reversible?
  The resolution here is structural: entropy increases not because of
  irreversible microscopic dynamics but because the visited-cell set is a
  monotone accumulator.
  Running time backward does not un-visit cells.
  The arrow of time is a consequence of the partition's set-theoretic
  structure, not of any dynamical asymmetry.
\end{remark}

\section{Mass as Non-Actualisation}
\label{sec:ch7-mass}

\begin{consequence}{Mass as Accumulated Non-Actualisation}{massmemory}
  The rest mass $m_0$ is the normalised rate at which the oscillator fails
  to actualise all partition cells per oscillation cycle:
  \begin{equation}
    m_0 = \frac{\hbar\omega_0}{c^2}
        = \frac{\hbar}{c^2} \cdot \frac{|\mathcal{U}(t)|}{t},
  \end{equation}
  where $\mathcal{U}(t) = \mathcal{A}_\infty \setminus \mathcal{V}(t)$ is
  the set of un-visited (\emph{unactualised}) cells at time $t$.
\end{consequence}

This is the deepest physical interpretation of mass in the framework.
Mass is not a primitive property; it is the memory of inaccessibility.
A massless particle ($m_0 = 0$) visits all partition cells at the maximum rate;
a massive particle carries a history of cells not visited.

\section{$E = mc^2$}
\label{sec:ch7-emc2}

\begin{consequence}{Mass--Energy Equivalence}{emc2}
  The total energy of a bounded oscillator at rest is
  \begin{equation}
    E = m_0 c^2.
  \end{equation}
\end{consequence}

\begin{proof}
  Immediate from Consequence~\ref{cons:restmass}: $m_0 c^2 = \hbar\omega_0 = E_0$.
  The energy of the ground mode is the rest energy.
\end{proof}

\begin{remark}
  The formula $E = mc^2$ is usually presented as a consequence of special
  relativity.
  Here it is a consequence of the Koopman ground state energy being non-zero
  on a bounded domain (Consequence~\ref{cons:restfreq}) combined with the
  dispersion relation (Consequence~\ref{cons:dispersion}).
  Special relativity is not assumed; it is derived (Chapter~\ref{ch:lorentz}).
\end{remark}

\section{Inertial--Gravitational Mass Equivalence}
\label{sec:ch7-equiv}

\begin{consequence}{Inertial equals Gravitational Mass}{equivmass}
  The mass parameter $\mu$ appearing in the partition
  Lagrangian~\eqref{eq:partitionlagrangian} is identical to the rest mass $m_0$
  of Consequence~\ref{cons:restmass}.
  The inertial mass (resistance to acceleration) equals the gravitational mass
  (coupling to the partition depth gradient).
\end{consequence}

\begin{proof}[Proof sketch]
  The partition Lagrangian has a single mass parameter $\mu$, which plays two
  roles: kinetic term coefficient (inertia) and coupling to $\Depth$ (gravity).
  These are the same parameter by the uniqueness of the Lagrangian
  (Consequence~\ref{cons:partitionlagrangian}).
  The equivalence principle is a consequence of the Lagrangian's structure,
  not an independent postulate.
\end{proof}
