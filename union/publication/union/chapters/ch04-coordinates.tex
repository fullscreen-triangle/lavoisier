\noindent
The previous chapter established that bounded systems oscillate and that
every observable decomposes into countably many modes ($E=\hbar\omega$).
This chapter answers the next question: what labels the modes?

The answer emerges from the geometry and topology of the bounded region
itself.
A bounded region in three spatial dimensions, equipped with a nested
measurable partition (Axiom~\ref{ax:bpsl}(iii)), admits a canonical
coordinate system $(n,l,m,s)$ --- the principal depth, angular
complexity, orientation, and chirality.
These are not postulated.
They are the unique coordinate system forced by the partition structure,
$SO(3)$ geometry, and the fundamental group $\pi_{1}(SO(3))=\mathbb{Z}_{2}$.

% ── §4.1 ────────────────────────────────────────────────────────────────────
\section{Partition Algebra and Countable Generation}
\label{sec:ch4-partitionalgebra}

\begin{definition}[Partition cell, partition algebra]
\label{def:partition}
Under Axiom~\ref{ax:bpsl}(iii), a \emph{cell} of the partition
$\Partition_{n}$ is a maximal measurable subset $\omega\subseteq\Omega$
such that every refining partition $\Partition_{m}$ ($m\ge n$) contains
either $\omega$ itself or subsets of $\omega$.
The \emph{partition algebra} at depth $n$ is the Boolean algebra
$\mathcal{A}_{n}$ of finite unions of cells of $\Partition_{n}$.
\end{definition}

\begin{consequence}{Countable generation}{partitionalgebra}
Under Axiom~\ref{ax:bpsl}, the Borel $\sigma$-algebra $\mathcal{B}$ of
$\Omega$ is generated by the countable algebra
$\mathcal{A}_{\infty}=\bigcup_{n\ge 0}\mathcal{A}_{n}$, and every
observable $f\in L^{2}(\Omega,\mu)$ admits a countable partition-basis
expansion.
\end{consequence}

\begin{proof}
By Axiom~\ref{ax:bpsl}(iii), $\bigvee_{n\ge 0}\Partition_{n}=\mathcal{B}$
modulo $\mu$-null sets.
Each $\Partition_{n}$ contains finitely many cells: a partition of a
finite-measure set $\Omega$ into positive-measure cells has only finitely
many cells at each level.
Hence $\mathcal{A}_{\infty}=\bigcup_{n}\mathcal{A}_{n}$ is a countable
algebra generating $\mathcal{B}$.
That every $f\in L^{2}$ admits a countable partition-basis expansion is
the standard result that indicator functions of cells form a complete
orthogonal system~\citep{halmos1950,rudin1987}.
\end{proof}

\begin{consequence}{Finite-information categorical approximation}{categorical}
Let an observer have finite information capacity $I$ bits.
Then the observer can distinguish at most $2^{I}$ cells of any partition
of $\Omega$; every measurement outcome is a categorical assignment to one
of these cells.
\end{consequence}

\begin{proof}
An observer with $I$ bits has $2^{I}$ distinct internal states; by
Shannon's coding theorem~\citep{shannon1948} no measurement outcome can
carry more than $I$ bits of information.
The observer's reading is a measurable function from $\Omega$ to a set
of at most $2^{I}$ labels; it factors through a partition of $\Omega$
into at most $2^{I}$ cells~\citep{landauer1961}.
\end{proof}

\begin{remark}
Consequence~\ref{cons:categorical} is the axiomatic origin of the
\emph{quantisation of observation}: no measurement produces a continuum
of outcomes; continua exist in the unrealised phase-space background,
while realised measurements populate discrete cells.
\end{remark}

\begin{corollary}[Minimum measurement energy]
\label{cor:measurementenergy}
The minimum energy required to distinguish two cells at depth $n$ is
\begin{equation}
  E_{\min}(n) = \kB T_{\min}\ln 2,
  \label{eq:landauer}
\end{equation}
where $T_{\min}$ is the environmental temperature.
\end{corollary}

\begin{proof}
Landauer's bound~\citep{landauer1961}: any measurement distinguishing two
states requires logical erasure of at least one bit, costing
$\kB T\ln 2$ at temperature $T$.
\end{proof}

% ── §4.2 ────────────────────────────────────────────────────────────────────
\section{Partition Coordinates $(n,l,m,s)$}
\label{sec:ch4-quantumnumbers}

We derive the canonical labelling system for partition cells.
The derivation uses the partition structure of Axiom~\ref{ax:bpsl}(iii),
the geometry of $SO(3)$, and its topology via $\pi_{1}(SO(3))$.

\begin{definition}[Principal depth]
\label{def:principal}
For $x\in\Omega$, the \emph{principal depth} $n(x)$ is the index $n$ of
the coarsest partition $\Partition_{n}$ in which $x$ occupies a cell
of positive measure that is distinguished from its parent cell in
$\Partition_{n-1}$.
\end{definition}

\begin{lemma}
\label{lem:nispositive}
For every $x\in\Omega$, $n(x)\in\mathbb{N}^{+}=\{1,2,3,\ldots\}$.
\end{lemma}

\begin{proof}
By Axiom~\ref{ax:bpsl}(iii), $\Partition_{0}$ is the coarsest partition;
positive-measure cells exist at every level, so $n(x)\ge 1$, and
$n(x)\in\mathbb{N}^{+}$.
\end{proof}

\begin{definition}[Angular complexity]
\label{def:angular}
At principal depth $n$, the cell containing $x$ admits a geometric
decomposition into an $(n-1)$-dimensional radial fibration and a
two-dimensional angular profile on the $(n-1)$-sphere.
The \emph{angular complexity} $l(x)$ is the index of the spherical
harmonic representing the angular profile of the cell boundary.
\end{definition}

\begin{lemma}[Range constraint $l<n$]
\label{lem:langeconstraint}
At principal depth $n$, the angular complexity satisfies
$0\le l\le n-1$.
\end{lemma}

\begin{proof}
The total number of independent scalar indices in a three-dimensional
bounded region at nesting depth $n$ is $n$.
One index is the principal depth itself; the remaining $n-1$ split
between the radial node count $n_{r}\ge 0$ and the angular index
$l\ge 0$ via the Sturm--Liouville relation
$n_{r}+l=n-1$~\citep{arfkenweber2005}.
Since $n_{r}\ge 0$, we have $l\le n-1$; combined with $l\ge 0$,
this yields $0\le l\le n-1$.
\end{proof}

\begin{lemma}[Orientation index range]
\label{lem:mrange}
At angular complexity $l$, the orientation index $m$ takes values in
$\{-l,-l+1,\ldots,+l\}$, a total of $2l+1$ values.
\end{lemma}

\begin{proof}
The $l$th irreducible representation of $SO(3)$ has dimension
$2l+1$~\citep{wigner1959,sakurai2017}.
The partition at depth $(n,l)$ inherits an $SO(3)$ action from the
spherical symmetry of $\Omega$; decomposing into $SO(3)$-irreducibles
yields $2l+1$ angular sub-cells at each $(n,l)$.
\end{proof}

\begin{remark}[Zeeman observation]
The Zeeman effect~\citep{zeeman1897} places an atom in an external
magnetic field and splits a spectral line of angular quantum number $l$
into exactly $2l+1$ components.
The count is exact; no fractional or continuous splitting is ever
observed.
This is the direct experimental confirmation of Lemma~\ref{lem:mrange}.
\end{remark}

\begin{lemma}[Binary chirality]
\label{lem:chirality}
The partition at each $(n,l,m)$ admits a further binary distinction
$s\in\{-\tfrac{1}{2},+\tfrac{1}{2}\}$ corresponding to the two elements
of the fundamental group $\pi_{1}(SO(3))=\mathbb{Z}_{2}$.
\end{lemma}

\begin{proof}
$SO(3)$ is not simply connected; its fundamental group is
$\mathbb{Z}_{2}$~\citep{nakahara2003}.
The universal cover $SU(2)\to SO(3)$ is a two-sheeted covering, so each
$SO(3)$-orbit at level $(n,l,m)$ lifts to exactly two preimages in
$SU(2)$.
Label these by $s\in\{-\tfrac{1}{2},+\tfrac{1}{2}\}$; the binary
chirality coordinate $s$ is well-defined at every $(n,l,m)$.
\end{proof}

\begin{remark}[The integer--half-integer distinction]
The boson/fermion distinction is the distinction between zero and non-zero
classes in $\pi_{1}(SO(3))=\mathbb{Z}_{2}$.
A bosonic chirality index is $s=0\pmod{1}$ (trivial loop class); a
fermionic index is $s=\tfrac{1}{2}\pmod{1}$ (non-trivial loop class).
This topological origin explains the 720° return requirement for fermions:
a fermion wave function requires two full rotations to return to its
original state, because the fermion's loop class sits in the non-trivial
element of $\pi_{1}(SO(3))$, which requires the double cover $SU(2)$ for
continuous unwinding.
\end{remark}

\begin{consequence}{Partition coordinates}{quantumnumbers}
Every cell of the partition hierarchy at maximum depth is labelled by a
quadruple $(n,l,m,s)$ with
\begin{equation}
  n\in\mathbb{N}^{+},\quad
  l\in\{0,1,\ldots,n-1\},\quad
  m\in\{-l,\ldots,+l\},\quad
  s\in\{-\tfrac{1}{2},+\tfrac{1}{2}\}.
  \label{eq:nlms}
\end{equation}
\end{consequence}

\begin{proof}
Combine Lemmas~\ref{lem:nispositive}, \ref{lem:langeconstraint},
\ref{lem:mrange}, and~\ref{lem:chirality}.
\end{proof}

\begin{remark}
Consequence~\ref{cons:quantumnumbers} produces exactly the four quantum
numbers of atomic spectroscopy with their standard range constraints,
as observed in all atomic spectra~\citep{herzberg1944,nistasd}.
The derivation uses no postulate about ``orbital angular momentum'' or
``spin''; both emerge from the combined geometry and topology of $SO(3)$
acting on a bounded region.
\end{remark}

\section{Shell Capacity $C(n)=2n^{2}$}
\label{sec:ch4-shellcapacity}

\begin{consequence}{Shell capacity}{shellcapacity}
The number of distinct partition cells at principal depth $n$ is
\begin{equation}
  C(n) = 2n^{2}.
  \label{eq:shellcap}
\end{equation}
\end{consequence}

\begin{proof}
At depth $n$: $l$ takes values $0,1,\ldots,n-1$ (Lemma~\ref{lem:langeconstraint});
for each $l$, $m$ takes $2l+1$ values (Lemma~\ref{lem:mrange});
for each $(n,l,m)$, $s$ takes 2 values (Lemma~\ref{lem:chirality}).
Hence
\begin{equation*}
  C(n) = \sum_{l=0}^{n-1}(2l+1)\cdot 2
       = 2\sum_{l=0}^{n-1}(2l+1).
\end{equation*}
Using the standard identity $\sum_{l=0}^{n-1}(2l+1)=n^{2}$ (the first
$n$ odd numbers sum to $n^{2}$):
$C(n)=2n^{2}$.
\end{proof}

\begin{example}
For $n=1,2,3,4,5$, equation~\eqref{eq:shellcap} gives
$C(n)=2,8,18,32,50$, matching the observed maximum occupation of the
$K,L,M,N,O$ electron shells in the periodic table~\citep{herzberg1944}.
\end{example}

\begin{consequence}{Sub-shell cell counts}{subshellcounts}
At $(n,l)$, the number of cells is $2(2l+1)$.
\end{consequence}

\begin{proof}
For fixed $(n,l)$: $m$ takes $2l+1$ values; $s$ takes 2 values.
Product: $2(2l+1)$.
\end{proof}

\begin{example}
The sub-shell capacities are $1s{:}\,2$; $2s{:}\,2$, $2p{:}\,6$;
$3s{:}\,2$, $3p{:}\,6$, $3d{:}\,10$; $4s{:}\,2$, $4p{:}\,6$,
$4d{:}\,10$, $4f{:}\,14$.
These match the slot counts of the standard periodic table.
\end{example}

\section{Selection Rules}
\label{sec:ch4-selectionrules}

\begin{consequence}{Selection rules}{selectionrules}
Under continuous oscillatory boundary deformation, transitions between
partition cells $(n,l,m,s)\to(n',l',m',s')$ are allowed only if
\begin{equation}
  \Delta l = \pm 1,\qquad
  \Delta m \in \{-1,0,+1\},\qquad
  \Delta s = 0.
  \label{eq:selectrules}
\end{equation}
\end{consequence}

\begin{proof}
\textbf{Step 1} ($\Delta l=\pm 1$).
Continuous oscillatory boundary deformation couples to spherical harmonics
$Y_{1}^{q}$ (a rank-1 tensor operator under $SO(3)$).
By the Wigner--Eckart theorem~\citep{wigner1959,sakurai2017}, matrix
elements of a rank-$K$ tensor between representations $l$ and $l'$ vanish
unless $|l-l'|\le K\le l+l'$.
For $K=1$: $|\Delta l|\le 1$.
Parity under inversion excludes $\Delta l=0$ (since $Y_{1}^{q}$ has parity
$-1$, coupling states of opposite parity).
Hence $\Delta l=\pm 1$.

\textbf{Step 2} ($\Delta m\in\{-1,0,+1\}$).
The rank-1 tensor has components $q\in\{-1,0,+1\}$; the Wigner--Eckart
theorem gives $\Delta m=q\in\{-1,0,+1\}$.

\textbf{Step 3} ($\Delta s=0$).
The chirality $s$ is the topological label of an element of
$\pi_{1}(SO(3))=\mathbb{Z}_{2}$ (Lemma~\ref{lem:chirality}).
A continuous boundary deformation is a homotopy; homotopies preserve
elements of the fundamental group.
Hence $\Delta s=0$.
\end{proof}

\begin{remark}
The selection rules~\eqref{eq:selectrules} are those observed in all
atomic electric dipole transitions~\citep{herzberg1944,nistasd}.
No additional postulate (such as ``only electric-dipole radiation is
allowed'') is introduced; the restriction to rank-1 tensors emerges from
the continuity of boundary deformations implicit in
Axiom~\ref{ax:bpsl}(iii).
\end{remark}

\begin{corollary}[Parity conservation]
\label{cor:parity}
Electric dipole transitions require a change of parity:
$\Delta l=\pm 1$ connects states of opposite parity $(-1)^{l}$.
\end{corollary}

\begin{corollary}[Forbidden transitions]
\label{cor:forbidden}
Transitions with $\Delta l=0$ are forbidden at electric-dipole order;
they are permitted only at higher multipole order (electric quadrupole,
magnetic dipole), suppressed by $\sim(a_{0}/\lambda)^{2}$.
Observed rates for forbidden lines are suppressed by $10^{-6}$ to
$10^{-10}$ relative to allowed lines~\citep{herzberg1944}, consistent
with Consequence~\ref{cons:selectionrules}.
\end{corollary}

\section{The Exclusion Principle}
\label{sec:ch4-pauli}

\begin{consequence}{Exclusion}{pauli}
Let two indistinguishable constituents occupy a common partition hierarchy
of Axiom~\ref{ax:bpsl}(iii).
Then no two constituents can be assigned identical quadruples $(n,l,m,s)$.
\end{consequence}

\begin{proof}
We give two independent proofs.

\textbf{Proof (a) --- Identity of Indiscernibles.}
Suppose two constituents share $(n,l,m,s)$.
By Consequence~\ref{cons:quantumnumbers}, this labels a single partition
cell at maximum depth; by Axiom~\ref{ax:bpsl}(iii), every measurable
observable on $\Omega$ is $\mu$-a.e.\ determined by the cell.
Hence the two constituents are indistinguishable by any measurement.
But the cell count is an observable (the trace of a partition indicator);
two constituents in the same cell is the same as one constituent by the
indistinguishability, contradicting the standing assumption of two
constituents.

\textbf{Proof (b) --- Compression-cost divergence.}
The volume $\mathcal{V}$ of a cell at $(n,l,m,s)$ is a fixed positive
number (Axiom~\ref{ax:bpsl}(iii)).
The structural information cost of distinguishing two indistinguishable
objects in identical cells is $\kB T\ln 2$ per distinction
(Corollary~\ref{cor:measurementenergy}).
This cost is independent of temperature in the zero-$T$ limit, where only
structural information remains.
The structural information deficit forbids double occupancy.
\end{proof}

\begin{proposition}[Spin--statistics correspondence]
\label{prop:spinstatistics}
Constituents with half-integer chirality ($s=\pm\tfrac{1}{2}$) occupy
antisymmetric cells; constituents with integer chirality occupy symmetric
cells.
\end{proposition}

\begin{proof}
The chirality $s$ is the element of $\pi_{1}(SO(3))=\mathbb{Z}_{2}$
identifying the boundary homotopy class of the constituent.
Two constituents undergoing combined exchange traverse a path of class
$s+s=2s\pmod{2}$.
For $s=\tfrac{1}{2}$: $2s=1\pmod{2}$, a non-trivial loop, so the
two-constituent wave function acquires a minus sign under exchange
(antisymmetric).
For $s=1$: $2s=0\pmod{2}$, the trivial loop, so the sign is positive
(symmetric).
This reproduces the Pauli spin-statistics rule~\citep{pauli1925}.
\end{proof}

\begin{remark}[Counterfactual]
If the exclusion principle did not hold, all electrons would fall into the
$1s$ ground state of any atom, and all atoms would have the same chemistry
as hydrogen-like ions.
The periodic table --- with its structure of repeating periods, transition
metals, and rare earths --- would not exist.
The observed structure of the periodic table is the direct empirical
confirmation of Consequence~\ref{cons:pauli}.
\end{remark}

\section{Compression Cost and Binding Energy}
\label{sec:ch4-compression}

\begin{consequence}{Compression cost}{compression}
Let $N$ constituents occupy a volume $\mathcal{V}$ in configuration
space, with individual volumes $\mathcal{V}_{0}$.
The free energy of compression is
\begin{equation}
  \mathcal{E}(N,\mathcal{V})
  = N\kB T\,\ln\!\frac{N\mathcal{V}_{0}}{\mathcal{V}}.
  \label{eq:compressioncost}
\end{equation}
As $\mathcal{V}\to 0$, $\mathcal{E}\to\infty$.
\end{consequence}

\begin{proof}
The categorical phase-space volume per constituent at temperature $T$ is
$\mathcal{V}_{0}$ (an individual cell of the partition at the
energy-appropriate depth).
The free energy of confining $N$ independent Boltzmann-distributed
constituents to joint volume $\mathcal{V}$ relative to their separate
volume $N\mathcal{V}_{0}$ is, by standard
thermodynamics~\citep{landaulifshitz5},
\begin{equation*}
  \mathcal{E}
  = -T\Delta S
  = -T\cdot N\kB\ln\!\frac{\mathcal{V}}{N\mathcal{V}_{0}}
  = N\kB T\ln\!\frac{N\mathcal{V}_{0}}{\mathcal{V}},
\end{equation*}
which is~\eqref{eq:compressioncost}.
\end{proof}

\begin{remark}
Equation~\eqref{eq:compressioncost} is the thermodynamic underpinning of
the exclusion principle and the rigidity of matter against collapse.
At nuclear-scale compression, the required energy exceeds the available
rest energy; the configuration collapses to a different sub-structure
(electron capture, neutron formation).
The compression cost is the lower bound for gravitational collapse and
the origin of neutron-star degeneracy pressure.
\end{remark}

\begin{consequence}{Sub-additivity of partition depth (binding)}{bindingenergy}
For a composite $\mathcal{C}$ of $N$ bound constituents with individual
partition depths $\Depth_{i}$, the composite depth satisfies
$\Depth(\mathcal{C})<\sum_{i}\Depth_{i}$, and the binding energy is
\begin{equation}
  E_{\mathrm{bind}}
  = \kB T\ln b\cdot\Delta\Depth,
  \qquad
  \Delta\Depth := \sum_{i}\Depth_{i}-\Depth(\mathcal{C}),
  \label{eq:bindingenergy}
\end{equation}
where $b$ is the branching base of the partition tree.
\end{consequence}

\begin{proof}
In a bound state, constituents share common partition boundaries.
The partition cells describing relative positions of constituent pairs
collapse under the sharing of the composite envelope, reducing the total
distinguishable cell count below the unshared total: strictly,
$\Depth(\mathcal{C})<\sum_{i}\Depth_{i}$.
By Consequence~\ref{cons:compression}, reducing the partition depth by
$\Delta\Depth$ frees free energy
$\kB T\ln(b^{\Delta\Depth})=\kB T\ln b\cdot\Delta\Depth$.
This released energy is the binding energy.
\end{proof}

\begin{example}[Hydrogen ionisation energy]
For hydrogen at discharge-lamp temperature ($\kB T\sim 1\ \mathrm{eV}$),
the partition depth reduction upon binding an electron to a proton in the
ground state is $\Delta\Depth\approx 13$ natural-log units, giving
$E_{\mathrm{bind}}\approx 13\ \mathrm{eV}$, consistent with the measured
ionisation energy of $13.598\ \mathrm{eV}$~\citep{nistasd}.
\end{example}

\begin{example}[H$_{2}$ molecular binding]
The H$_{2}$ dissociation energy is $4.48\ \mathrm{eV}$.
The partition depth reduction from two separated hydrogen atoms to one
molecular envelope is $\Delta\Depth\approx 4.5$ natural-log units,
giving $E_{\mathrm{bind}}\approx 4.5\ \mathrm{eV}$, consistent with
measurement.
\end{example}

\section{Three-Dimensional Space from Partition Geometry}
\label{sec:ch4-spaceemergence}

\begin{consequence}{Emergence of three-dimensional space}{spaceemergence}
The partition coordinates of Consequence~\ref{cons:quantumnumbers} endow
every bounded phase space with a natural three-dimensional spatial
structure.
The dimension $d=3$ is the unique integer for which both the angular
coordinate $(l,m)$ and the chirality coordinate $s$ are non-trivial and
generate representations of distinct fundamental groups.
\end{consequence}

\begin{proof}
Consequence~\ref{cons:quantumnumbers} established $(n,l,m,s)$ without
assuming any embedding space; the dimension of the emergent space is
forced by requiring the orientation coordinate $m$ to be integer-valued
and the chirality coordinate $s$ to be binary.

For $d=2$: $\pi_{1}(SO(2))=\mathbb{Z}$; the chirality is continuous,
not binary.

For $d=3$: $\pi_{1}(SO(3))=\mathbb{Z}_{2}$; chirality is exactly binary.
The $SO(3)$-irreducible count $2l+1$ matches the integer-indexed
angular coordinate $l$.

For $d\ge 4$: $\pi_{1}(SO(d))=\mathbb{Z}_{2}$ as well, but the
orientation sub-representation space grows as $d(d-1)/2$ rather than
$2l+1$, losing the one-to-one match with the integer coordinate $l$.

Thus $d=3$ is uniquely distinguished: the smallest dimension admitting
binary chirality and the unique dimension where the orientation count
$2l+1$ matches the irreducible representations of $SO(d)$.
\end{proof}

\begin{corollary}
\label{cor:spatialreps}
The angular coordinates $(l,m)$ generate spherical harmonics
$Y_{l}^{m}(\theta,\varphi)$ as natural basis functions on the two-sphere.
Radial extension is provided by the principal coordinate $n$ via the
Bohr-radius scaling $r\propto n^{2}$ (Consequence~\ref{cons:aufbau}).
Three-dimensional Euclidean space is a derived object from the partition
structure, not an independent postulate.
\end{corollary}

\section{Economy of Postulates}
\label{sec:ch4-economy}

The derivations of this chapter replace the following independent postulates
of standard quantum mechanics:

\begin{enumerate}
  \item \emph{Quantum numbers as postulated labels}: In standard QM, the
    quadruple $(n,l,m,s)$ is introduced by solving the Schrödinger equation
    with boundary conditions.
    Here, it is the canonical coordinate system forced by the partition
    geometry of any bounded three-dimensional system
    (Consequence~\ref{cons:quantumnumbers}).

  \item \emph{Shell capacity as experimental fact}: The atomic capacity
    $2n^{2}$ is observed and tabulated.
    Here, it is derived algebraically from the sum $\sum_{l=0}^{n-1}(2l+1)=n^{2}$
    (Consequence~\ref{cons:shellcapacity}).

  \item \emph{Selection rules as postulated}: In standard QM, electric
    dipole selection rules are postulated via the Wigner--Eckart theorem
    applied to the electromagnetic coupling.
    Here, the rank-1 coupling emerges from the continuity of boundary
    deformations in the partition hierarchy
    (Consequence~\ref{cons:selectionrules}).

  \item \emph{Pauli exclusion as postulate}: Historically introduced as an
    independent postulate by Pauli in 1925~\citep{pauli1925}.
    Here, it follows from either the Identity of Indiscernibles or the
    compression-cost divergence in the partition cell
    (Consequence~\ref{cons:pauli}).
\end{enumerate}

                  title={\textbf{Chapter 4 Summary}}]
\begin{itemize}[noitemsep]
  \item \textbf{Consequences \ref{cons:partitionalgebra}--\ref{cons:categorical}}:
    The Borel $\sigma$-algebra is countably generated by the partition;
    every measurement outcome is a categorical cell assignment.
  \item \textbf{Consequence \ref{cons:quantumnumbers}} (Partition

    theory, and $\pi_{1}(SO(3))=\mathbb{Z}_{2}$.
  \item \textbf{Consequence \ref{cons:shellcapacity}} (Shell capacity):
    $C(n)=2n^{2}$, matching the $K,L,M,N,\ldots$ shell occupancies of the

  \item \textbf{Consequence \ref{cons:selectionrules}} (Selection rules):

  \item \textbf{Consequence \ref{cons:pauli}} (Exclusion): No two

  \item \textbf{Consequence \ref{cons:bindingenergy}} (Binding energy):
    $E_{\mathrm{bind}}=\kB T\ln b\cdot\Delta\Depth$, reproducing hydrogen
    ionisation ($13.6\ \mathrm{eV}$) and molecular dissociation energies.
\end{itemize}
