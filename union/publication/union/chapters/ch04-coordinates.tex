\section{The Partition Algebra}
\label{sec:ch4-algebra}

Clause~(iii) of BPSL guarantees a sequence of refining measurable partitions
$\Partition_1 \prec \Partition_2 \prec \cdots$ whose union generates the
full $\sigma$-algebra.
The \emph{partition algebra} is
\begin{equation}
  \mathcal{A}_\infty = \bigvee_{n \geq 1} \mathcal{A}(\Partition_n).
\end{equation}

\begin{consequence}{Countably Generated Partition Algebra}{partitionalgebra}
  $\mathcal{A}_\infty$ is a countably generated Boolean algebra dense in
  $\mathcal{B}(\Omega)$.
\end{consequence}

\section{Categorical Observation}
\label{sec:ch4-categorical}

\begin{definition}[Categorical observation]
  A \emph{categorical observation at depth $n$} is the identification of the
  unique cell $A \in \Partition_n$ containing the current state of the system.
\end{definition}

\begin{consequence}{Categorical Observation}{categorical}
  Any observer with finite information capacity $I < \infty$ bits can only
  perform categorical observations at some finite depth $n^* =
  \lfloor I / H(\Partition_1) \rfloor$.
\end{consequence}

\begin{proof}[Proof sketch]
  By the Shannon coding theorem, a message identifying one cell of
  $\Partition_n$ requires $H(\Partition_n)$ bits.
  An observer with $I$ bits cannot resolve partitions deeper than depth
  $n^*$ where $H(\Partition_{n^*}) \leq I < H(\Partition_{n^*+1})$.
\end{proof}

\section{Partition Coordinates from Geometry}
\label{sec:ch4-quantumnumbers}

We now derive the four quantum numbers from the geometry of the nested partition.

\begin{consequence}{Partition Coordinates $(n, l, m, s)$}{quantumnumbers}
  The cells of $\Partition_n$ are labelled by four coordinates:
  \begin{align}
    n &\in \mathbb{N}^+ && \text{(principal depth: nesting level)}, \\
    l &\in \{0, 1, \ldots, n-1\} && \text{(angular complexity: $SO(3)$ irrep label)}, \\
    m &\in \{-l, \ldots, l\} && \text{(orientation: $SO(3)$ weight)}, \\
    s &= \pm\tfrac{1}{2} && \text{(chirality: $\pi_1(SO(3)) \cong \mathbb{Z}/2$)}.
  \end{align}
\end{consequence}

\begin{proof}[Proof sketch — four Lemmas]
  \textit{Lemma ($n$ is a positive integer)}: the nesting depth of a measurable
  partition is a positive integer by definition.

  \textit{Lemma ($l < n$)}: at depth $n$, the number of angular sub-partitions
  cannot exceed $n$ without violating the hierarchical structure; the constraint
  $l < n$ follows from the dimension count of the $SO(3)$ action on the
  phase-space sphere.

  \textit{Lemma ($|m| \leq l$)}: the weight of an $SO(3)$ irreducible
  representation of dimension $2l+1$ satisfies $m \in \{-l, \ldots, l\}$.

  \textit{Lemma (chirality $s = \pm\half$)}: the partition hierarchy embeds in
  a simply connected covering space; the fundamental group
  $\pi_1(SO(3)) \cong \mathbb{Z}/2\mathbb{Z}$ generates a $\mathbb{Z}/2$-valued
  topological invariant, which we identify as spin $s = \pm\half$.
\end{proof}

The four partition coordinates $(n, l, m, s)$ are identical to the four quantum
numbers of atomic physics.
They have been derived from the geometry of a bounded, hierarchically partitioned
phase space --- not postulated from spectroscopic observations.

\section{The Shell Capacity $C(n) = 2n^2$}
\label{sec:ch4-shellcapacity}

\begin{consequence}{Shell Capacity}{shellcapacity}
  The number of distinct partition cells at principal depth $n$ is
  \begin{equation}\label{eq:shellcapacity}
    C(n) = 2n^2.
  \end{equation}
\end{consequence}

\begin{proof}
  At depth $n$, sum over all allowed $(l, m, s)$:
  \begin{equation}
    C(n) = \sum_{l=0}^{n-1} \sum_{m=-l}^{l} 2
         = 2\sum_{l=0}^{n-1}(2l+1)
         = 2n^2.
  \end{equation}
  The factor of 2 counts the two chiralities $s = \pm\half$.
\end{proof}

The shell capacity $C(n) = 2, 8, 18, 32, \ldots$ for $n = 1, 2, 3, 4, \ldots$
is the exact shell filling capacity of atomic electron shells.
It is a theorem of partition geometry, not a spectroscopic accident.

\section{Selection Rules}
\label{sec:ch4-selection}

\begin{consequence}{Selection Rules}{selectionrules}
  Physical transitions between partition cells at depths $n, l, m$ and
  $n', l', m'$ satisfy:
  \begin{equation}
    \Delta l = \pm 1, \qquad \Delta m \in \{-1, 0, +1\}, \qquad
    \Delta s = 0.
  \end{equation}
\end{consequence}

\begin{proof}[Proof sketch]
  Transitions between partition cells correspond to matrix elements of
  a rank-1 spherical tensor (dipole operator) in the Koopman Hilbert space.
  The Wigner--Eckart theorem constrains non-zero matrix elements to
  $|\Delta l| = 1$, $|\Delta m| \leq 1$.
  The homotopy invariance of spin suppresses $\Delta s \neq 0$ for electric
  dipole transitions.
\end{proof}

\section{The Exclusion Principle}
\label{sec:ch4-pauli}

\begin{consequence}{Exclusion Principle}{pauli}
  No two indistinguishable fermions can occupy the same partition cell
  $(n, l, m, s)$.
\end{consequence}

\begin{proof}[Proof sketch]
  Each partition cell is a distinct measurable set with positive Liouville
  measure.
  Two indistinguishable particles occupying the same cell would collapse
  the distinguishable partition, reducing the effective partition depth by 1
  and violating the measure-theoretic separating property of
  $\mathcal{A}_\infty$.
  The minimum information cost to distinguish two particles in the same
  cell is infinite (they are identical).
  The Shannon entropy of the collapsed partition is ill-defined.
  Therefore, the partition algebra requires distinct cells for distinct
  fermion occupancies.
\end{proof}

\section{Compression Cost and Binding Energy}
\label{sec:ch4-binding}

\begin{consequence}{Compression Cost}{compression}
  Localising a system to a sub-cell of volume $V_{n+1} \subset V_n$ at depth
  $n+1$ costs entropy $\Delta S = \kB \ln(V_n / V_{n+1})$.
\end{consequence}

\begin{consequence}{Binding Energy}{bindingenergy}
  The energy released when two sub-systems combine at partition depth $n$ into
  a joint cell at depth $n+1$ is
  \begin{equation}
    E_\mathrm{bind} = \kB T\, \Delta S_\mathrm{joint}
                    < \kB T\, (\Delta S_1 + \Delta S_2),
  \end{equation}
  where the inequality follows from the sub-additivity of partition entropy.
  The binding energy is the entropic benefit of sharing a larger cell.
\end{consequence}
