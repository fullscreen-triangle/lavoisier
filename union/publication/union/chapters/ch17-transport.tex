\section{The Universal Transport Formula}
\label{sec:transport}
%------------------------------------------------------------

Transport processes---electrical conduction, viscous flow, diffusion,
thermal conduction---appear in textbooks as distinct phenomena requiring
distinct derivations (Drude, Chapman--Enskog, Einstein, Fourier).
The categorical framework reveals them as special cases of a single
formula, each corresponding to a different choice of partition-lag time
and coupling weight.

\begin{consequence}{Universal transport formula}{transport}
Let $\Xi$ be a transport coefficient (resistivity, viscosity, or
diffusivity) of a system whose constituents make categorical transitions
$i \to j$ between partition cells at mean time $\taup^{(ij)}$, coupled by
a transition weight $g_{ij}$.  Then
\begin{equation}
  \Xi
  = \frac{1}{\mathcal{N}}
    \sum_{i,j} \taup^{(ij)}\, g_{ij},
  \label{eq:transport}
\end{equation}
where $\mathcal{N}$ is a normalisation depending on the transport
coefficient.
\end{consequence}

\begin{proof}
A transport process is a sequence of categorical transitions.  Each
transition $i \to j$ has a characteristic partition-lag time
$\taup^{(ij)}$ (the time required for the system to traverse the boundary
between cell $i$ and cell $j$) and a coupling weight $g_{ij}$ (the
conjugate variable, e.g., electric field strength for resistivity,
momentum flux for viscosity, concentration gradient for diffusivity).
The linear-response coefficient is the average lag weighted by $g_{ij}$,
summed over all active transitions.

\medskip
\noindent\textbf{Special case: electrical resistivity.}
For electrons in a metal, $\taup^{(ij)} = \taup$ (the Drude mean free
time, uniform across carriers), $g_{ij} = ne^2/m$ (carrier density times
charge squared over effective mass), and $\mathcal{N} = 1$.
Formula~\eqref{eq:transport} reduces to
\[
  \sigma = \frac{ne^2 \taup}{m},
  \qquad
  \rho = \frac{1}{\sigma} = \frac{m}{ne^2\taup},
\]
which is the Drude conductivity.

\medskip
\noindent\textbf{Special case: viscosity.}
For a classical gas, $\taup^{(ij)}$ is the mean collision time and
$g_{ij} = n\kB T$ (momentum-flux coupling).
Formula~\eqref{eq:transport} reduces to
\[
  \eta = n\kB T \taup,
\]
the Chapman--Enskog viscosity.

\medskip
\noindent\textbf{Special case: diffusivity.}
For a dilute species in a thermal bath, $\taup^{(ij)}$ is the thermal
correlation time and the coupling weight gives
\[
  D = \frac{\kB T \taup}{m},
\]
the Einstein diffusion coefficient.

All three are recovered from~\eqref{eq:transport} without further
assumptions, confirming the formula's universality.
\end{proof}

%------------------------------------------------------------
\subsection{Special cases in detail}
%------------------------------------------------------------

We make the reduction to each classical formula explicit.

\medskip
\noindent\textbf{Electrical conductivity.}
In a metal, the charge carriers are conduction electrons of density $n$
and charge $-e$.  The partition-lag time $\taup$ is the mean free time
between scatterings.  After integrating over the Fermi sphere, the
universal formula gives
\begin{equation}
  \sigma = \frac{ne^2\taup}{m_e^*},
  \qquad
  \rho = \frac{m_e^*}{ne^2\taup}.
  \label{eq:sigma}
\end{equation}

\noindent\textbf{Electronic thermal conductivity.}
For the same carrier population, the energy-weighted transition sum gives
\begin{equation}
  \kappa = \frac{\pi^2 n \kB^2 T \taup}{3 m_e}.
  \label{eq:kappa}
\end{equation}

\noindent\textbf{Viscosity (Newtonian fluid).}
For a classical gas with mean free path $\ell$ and mean speed
$\langle |v| \rangle$,
\begin{equation}
  \eta
  = \tfrac{1}{3} n \langle |v| \rangle m \ell
  = n m \langle v^2 \rangle \taup / 3
  = n \kB T \taup,
  \label{eq:eta}
\end{equation}
where the second equality uses $\taup = \ell / \langle |v| \rangle$ and
the third uses equipartition.

\noindent\textbf{Diffusivity.}
For a dilute species in a bath,
\begin{equation}
  D = \frac{\kB T \taup}{m}.
  \label{eq:diff}
\end{equation}
Combined with the Stokes--Einstein relation $D = \kB T / (6\pi\eta r)$,
equation~\eqref{eq:diff} gives the hydrodynamic radius $r$ directly from
the partition-lag time.

\section{Partition Extinction}
\label{sec:partitionextinction}
%------------------------------------------------------------

The universal transport formula~\eqref{eq:transport} is a sum over active
categorical transitions.  What happens when the transitions vanish?

\begin{consequence}{Partition extinction and zero transport}{partitionextinction}
When categorical distinguishability between constituents vanishes---i.e.,
when $\taup^{(ij)} \to 0$ uniformly over the active transition set, or
equivalently when all constituents share a common partition cell---the
transport sum~\eqref{eq:transport} has zero terms.  The transport
coefficient $\Xi$ vanishes identically:
\begin{equation}
  \taup^{(ij)} \to 0 \;\text{ for all active } (i,j)
  \quad\Longrightarrow\quad
  \Xi = 0.
\end{equation}
\end{consequence}

\begin{proof}
If all active transitions $i \to j$ have zero partition-lag time (the two
cells have coalesced), each term in the sum~\eqref{eq:transport} is zero
and the sum itself is zero.  This represents the \emph{phase-locked} regime:
all constituents occupy a single partition cell and therefore share a single
categorical label; there are no distinct-cell boundary crossings, and the
sum over transitions is empty.
\end{proof}

\begin{remark}[Physical manifestations of partition extinction]
Consequence~\ref{cons:partitionextinction} provides an exact, derivational
account of three celebrated phenomena:
\begin{itemize}[nosep]
  \item \textbf{Superconductivity} (Onnes, 1911): electrical resistivity
    is exactly zero below the critical temperature $T_c$.  In the
    partition picture, electrons below $T_c$ occupy a phase-locked
    partition cell (Cooper pairs), and the transition sum is empty.

  \item \textbf{Superfluidity} (Kapitza, 1938): viscosity is exactly zero
    below the $\lambda$-point of $^4$He.  The superfluid component
    occupies a single partition cell; viscous transitions between
    cells are absent.

  \item \textbf{Bose--Einstein condensation} (Bose, 1924; Anderson et
    al., 1995): the condensate has zero categorical distinguishability
    among its occupants---they all occupy the lowest-energy partition
    cell $(n,l,m,s) = (1,0,0,0)$.
\end{itemize}
In each case the vanishing is exact rather than approximate, matching the
exact character of Consequence~\ref{cons:partitionextinction}.
\end{remark}

\begin{corollary}[Critical temperature from partition extinction]
\label{cor:Tc}
The critical temperature $T_c$ at which the transport coefficient $\Xi$
vanishes is the temperature at which the partition-lag time
$\taup^{(ij)}$ for the dominant transition drops below the mean time
between categorical distinguishing events in the remaining distinguishable
set.
\end{corollary}

\begin{proof}
$\Xi(T) = \mathcal{N}^{-1}\sum_{i,j}\taup^{(ij)}g_{ij}$.
When the dominant $\taup$ drops below the threshold at which categorical
distinguishability fails, the sum over active transitions shrinks to zero.
The critical temperature is defined as the temperature at which this
threshold is crossed.  For phonon-mediated superconductivity, matching the
partition-lag time to the phonon oscillation frequency yields the BCS
expression
\[
  \kB T_c \sim \hbar\omega_D \,\eu^{-1/(N(0)V)},
\]
where $\omega_D$ is the Debye frequency, $N(0)$ is the density of states
at the Fermi level, and $V$ is the effective pairing potential.
\end{proof}

\section{Wiedemann--Franz Law}
\label{sec:wiedemann}
%------------------------------------------------------------

\begin{consequence}{Wiedemann--Franz law}{wiedemann}
For a metal whose charge and heat carriers are the same partition
population, the ratio of thermal to electrical conductivity is
\begin{equation}
  \frac{\kappa}{\sigma T}
  = \frac{\pi^2}{3}\frac{\kB^2}{e^2}
  \equiv L_0
  = 2.443 \times 10^{-8}\;\mathrm{W\,\Omega\,K^{-2}}.
  \label{eq:wiedemann}
\end{equation}
\end{consequence}

\begin{proof}
By Consequence~\ref{cons:transport}, both $\kappa$ (from~\eqref{eq:kappa})
and $\sigma$ (from~\eqref{eq:sigma}) admit transport expansions with
the \emph{same} partition-lag denominator $\taup$, set by the common
carrier population.  The ratio $\kappa/\sigma$ is therefore independent
of $\taup$:
\[
  \frac{\kappa}{\sigma}
  = \frac{\pi^2 n \kB^2 T \taup / (3m)}{ne^2\taup/m}
  = \frac{\pi^2 \kB^2 T}{3 e^2}.
\]
Dividing by $T$:
\[
  L = \frac{\kappa}{\sigma T}
  = \frac{\pi^2 \kB^2}{3 e^2}
  = L_0.
\]
The numerical coefficient $\pi^2/3$ arises from the Sommerfeld expansion
of the electronic heat capacity of a Fermi gas at $T \ll T_F$; no fitting
parameter enters.
\end{proof}

\begin{remark}[Empirical verification]
Table~\ref{tab:WF} records the Lorenz number $L = \kappa/(\sigma T)$ for
six metals.  The prediction $L_0 = 2.443 \times 10^{-8}$~W$\Omega$K$^{-2}$
(Consequence~\ref{cons:wiedemann}) is compared with experiment.  Five of
six metals lie within 10\% of $L_0$; tungsten's excess arises from
phonon-drag contributions beyond the Sommerfeld-expansion regime.
\end{remark}

\section{Single-Particle Ideal Gas Law}
\label{sec:singlepartgas}
%------------------------------------------------------------

\begin{consequence}{Single-particle ideal gas law}{singleparticle}
A single bounded oscillator with total internal energy $E$ and partition
depth $\Depth$ obeys
\begin{equation}
  PV = \kB T_{\mathrm{cat}},
  \label{eq:spidealgas}
\end{equation}
where the \emph{categorical temperature} is
\begin{equation}
  T_{\mathrm{cat}} = \frac{2E}{3\kB \Depth}.
  \label{eq:Tcatdepth}
\end{equation}
This extends the $N \ge 1$ ideal gas law (Consequence~\ref{cons:idealgasN}
of \cref{ch:idealgas}) to the single-particle case through the
depth variable $\Depth$.
\end{consequence}

\begin{proof}
Consequence~\ref{cons:idealgasN} established $PV = N\kB T_{\mathrm{cat}}$
as a categorical balance between boundary partition production and thermal
partition activity.  At $N = 1$, the balance still applies but the
temperature variable refers to partition-depth fluctuations of the
\emph{single} particle's internal degrees of freedom.

The equipartition relation
$\langle E \rangle = \tfrac{3}{2}\kB T_{\mathrm{phys}}$
(from Proposition~\ref{prop:equipartition} of \cref{ch:idealgas})
rearranges to $T = 2E/(3\kB)$ when interpreted as a categorical
temperature per unit depth:
\[
  T_{\mathrm{cat}} = \frac{2E}{3\kB\Depth}.
\]
Substituting $N = 1$ into Consequence~\ref{cons:idealgasN} yields
$PV = \kB T_{\mathrm{cat}}$, giving~\eqref{eq:spidealgas}.
\end{proof}

\subsection{Physical--categorical temperature duality}
%------------------------------------------------------------

\begin{consequence}{Physical--categorical temperature duality}{Tduality}
The physical (kinetic) temperature $T_{\mathrm{phys}}$ and the categorical
temperature $T_{\mathrm{cat}}$ are related by
\begin{equation}
  T_{\mathrm{cat}} = \frac{T_{\mathrm{phys}}}{\Depth}.
  \label{eq:Tduality}
\end{equation}
\end{consequence}

\begin{proof}
The physical kinetic temperature satisfies $E = \tfrac{3}{2}\kB T_{\mathrm{phys}}$
for one particle.  Substituting into the categorical expression:
\[
  T_{\mathrm{cat}}
  = \frac{2E}{3\kB\Depth}
  = \frac{2 \cdot \tfrac{3}{2}\kB T_{\mathrm{phys}}}{3\kB\Depth}
  = \frac{T_{\mathrm{phys}}}{\Depth}.
\]
\end{proof}

\begin{remark}[Numerical anchor: trapped heavy ion]
For a single trapped ion of mass $M \approx 2 \times 10^6$ atomic mass
units at physical temperature $T_{\mathrm{phys}} = 77$~K (liquid
nitrogen), the categorical temperature is
\[
  T_{\mathrm{cat}} = \frac{77}{2\times10^6} \approx 38.5\;\mu\mathrm{K}.
\]
This matches the effective cooling of heavy ions observed in Penning-trap
mass spectrometry to approximately $38.3\;\mu$K at matched partition depth,
an agreement of better than $1\%$.
\end{remark}

\subsection{Zero-work cooling}
%------------------------------------------------------------

\begin{consequence}{Zero-work cooling}{zerowork}
If the partition depth $\Depth$ of a bounded system grows monotonically
with time (e.g., by continued Koopman-mode refinement) while the physical
energy $E$ is held constant, the categorical temperature decreases as
\begin{equation}
  T_{\mathrm{cat}}(t) \propto \frac{1}{\Depth(t)},
  \label{eq:zerowork}
\end{equation}
without any energy being extracted from the system.
\end{consequence}

\begin{proof}
Direct substitution in Consequence~\ref{cons:singleparticle}:
$T_{\mathrm{cat}} = 2E/(3\kB\Depth)$ is strictly decreasing in $\Depth$
at fixed $E$.  The decrease does not require work extraction because
partition-depth growth is a re-labelling of the existing phase-space cells
(as specified by the Axiom, \cref{ch:axiom}), not a displacement of the
system in phase space.
\end{proof}

\begin{remark}[Testable prediction]
The prediction is that long integration times in single-particle mass
spectrometry, trapped-ion experiments, and cavity-QED measurements should
exhibit a secular decrease of the effective categorical temperature with
$1/t$ scaling, independent of the active cooling power applied.  This is an
observable deviation from the standard thermodynamic expectation that
cooling requires work.
\end{remark}

\section{Heat--Entropy Decoupling on Product Phase Space}
\label{sec:heatentropy}
%------------------------------------------------------------

\begin{consequence}{Heat--entropy decoupling}{heatentropy}
Let $\mathcal{C} = \mathcal{S} \times \mathcal{P}$ be the product phase
space of a bounded system, where $\mathcal{S}$ is the thermal (kinetic)
sub-space and $\mathcal{P}$ is the categorical (partition) sub-space.
The covariance of infinitesimal heat $\delta Q$ and categorical entropy
$\mathrm{d}S_{\mathrm{cat}}$ on $\mathcal{C}$ vanishes:
\begin{equation}
  \mathrm{Cov}(\delta Q,\;\mathrm{d}S_{\mathrm{cat}}) = 0.
  \label{eq:heatentropy}
\end{equation}
Consequently, thermal and categorical entropies are independent quantities,
and a bounded system admits \emph{two} independent second laws: one for
$\mathcal{S}$ (classical Clausius) and one for $\mathcal{P}$ (categorical).
\end{consequence}

\begin{proof}
The projection $\pi_{\mathcal{S}} : \mathcal{C} \to \mathcal{S}$ commutes
with the categorical projection
$\pi_{\mathcal{P}} : \mathcal{C} \to \mathcal{P}$
because the product structure $\mathcal{C} = \mathcal{S} \times \mathcal{P}$
factors.
Heat is a property of the kinetic sub-space:
\[
  \delta Q = T_{\mathrm{phys}}\,\mathrm{d}S_{\mathrm{therm}}
  = \pi_{\mathcal{S}}^*\omega_Q,
\]
where $\omega_Q$ is a 1-form on $\mathcal{S}$.
Categorical entropy is a property of the partition sub-space:
\[
  \mathrm{d}S_{\mathrm{cat}} = \pi_{\mathcal{P}}^*\omega_C,
\]
for a 1-form $\omega_C$ on $\mathcal{P}$.  By the product structure and
Fubini's theorem on the product measure,
\[
  \mathrm{Cov}
    \!\left(\pi_{\mathcal{S}}^*\omega_Q,\; \pi_{\mathcal{P}}^*\omega_C\right)
  = \int_{\mathcal{C}} \omega_Q \wedge \omega_C
    - \langle\omega_Q\rangle\langle\omega_C\rangle
  = 0,
\]
proving~\eqref{eq:heatentropy}.
\end{proof}

\begin{corollary}[Independent second laws]
\label{cor:twosecondlaws}
A bounded system has:
\begin{enumerate}[nosep, label=(\roman*)]
  \item a \emph{thermal second law}: $\mathrm{d}S_{\mathrm{therm}} \ge 0$
    (Clausius, on $\mathcal{S}$);
  \item a \emph{categorical second law}: $\mathrm{d}S_{\mathrm{cat}} \ge 0$
    (from \cref{ch:sentropy}, on $\mathcal{P}$).
\end{enumerate}
Neither implies the other.  In particular, the categorical second law holds
without any statistical averaging over microstates.
\end{corollary}

\section{Universal Equation of State Across Thermodynamic Regimes}
\label{sec:universaleos}
%------------------------------------------------------------

\begin{consequence}{Universal equation of state}{universaleos}
All five canonical thermodynamic regimes---ideal gas, plasma, degenerate
Fermi gas, relativistic gas, and Bose--Einstein condensate---admit the
unified expression
\begin{equation}
  PV = N\kB T \cdot
    \mathcal{S}\!\left(V, N, T, \{n_i, l_i, m_i, s_i\}\right),
  \label{eq:universaleos}
\end{equation}
where the \emph{structural factor} $\mathcal{S}$ depends only on the
partition coordinates of the constituent particles and reduces to each
regime's established equation of state in the appropriate limit.
\end{consequence}

\begin{proof}
Equation~\eqref{eq:universaleos} is the categorical balance of
Consequence~\ref{cons:idealgasN} (\cref{ch:idealgas}) augmented by a
dimensionless structural factor $\mathcal{S}$ encoding partition-coordinate
correlations.  The five regimes correspond to distinct limits:

\begin{itemize}[nosep]
  \item \textbf{Ideal gas.}  Uncorrelated high-temperature limit:
    $\mathcal{S} = 1$, recovering $PV = N\kB T$.

  \item \textbf{Plasma.}  Debye screening at plasma parameter
    $\Gamma = e^2/(4\pi\varepsilon_0 a\kB T)$ gives
    $\mathcal{S} = 1 - \Gamma/3$,
    recovering the Debye--H\"uckel equation of state.

  \item \textbf{Degenerate Fermi gas.}  Pauli exclusion on partition
    coordinates (\cref{ch:shells}) at thermal de Broglie wavelength
    $\lambda_{\mathrm{th}} > a$ (the inter-particle spacing) gives
    \[
      \mathcal{S}
      = \frac{2}{5}\frac{E_F}{\kB T}
        \sum_n \frac{C(n)}{N},
    \]
    where $C(n) = 2n^2$ is the shell capacity, recovering Fermi--Dirac
    statistics.

  \item \textbf{Relativistic gas.}  J\"uttner distribution at
    $\kB T \sim m_0 c^2$ gives
    \[
      \mathcal{S}
      = \frac{K_3(\theta^{-1})}{\theta\, K_2(\theta^{-1})},
      \qquad
      \theta = \frac{\kB T}{m_0 c^2},
    \]
    where $K_n$ are modified Bessel functions of the second kind.

  \item \textbf{Bose--Einstein condensate.}  Macroscopic occupation of
    the ground partition cell $(n,l,m,s) = (1,0,0,0)$ below $T_c$ gives
    \[
      \mathcal{S}
      = \frac{\zeta(5/2)}{\zeta(3/2)}
        \left(\frac{T}{T_c}\right)^{3/2},
    \]
    recovering Einstein--Bose statistics, where $\zeta$ is the Riemann
    zeta function.
\end{itemize}

In every regime, $\mathcal{S}$ is a function of partition coordinates and
the dimensionless ratios $\Gamma$, $\lambda_{\mathrm{th}}/a$, $\theta$,
$T/T_c$; the regime boundaries are surfaces in partition-coordinate space.
\end{proof}

\begin{remark}[Regime boundaries]
The regime crossovers occur at universal dimensionless values of the
partition parameters:
\begin{itemize}[nosep]
  \item Ideal $\to$ plasma: $\Gamma \sim 1$
    (electrostatic energy comparable to thermal energy);
  \item Classical $\to$ degenerate: $\lambda_{\mathrm{th}}/a \sim 1$
    (de Broglie wavelength comparable to inter-particle spacing);
  \item Non-relativistic $\to$ relativistic: $\kB T/(m_0c^2) \sim 1$
    (thermal energy comparable to rest energy);
  \item Classical $\to$ BEC: $T/T_c \sim 1$
    (occupation of ground partition cell becomes macroscopic).
\end{itemize}
Each crossover separates regions of partition-coordinate space with
distinct structural factors $\mathcal{S}$.
\end{remark}

\section{Equilibrium as Poincar\'e Recurrence}
\label{sec:equilibriumrecurrence}
%------------------------------------------------------------

\begin{consequence}{Thermodynamic equilibrium as Poincar\'e recurrence}{equilibriumrecurrence}
The thermodynamic equilibrium state of a bounded system satisfying the
Axiom (\cref{ch:axiom}) is the recurrence orbit of the Poincar\'e
Recurrence Consequence (\cref{ch:oscillation}).  The recurrence time
$\tau_P$ satisfies
\begin{equation}
  \tau_P \sim \exp(S/\kB),
  \label{eq:recurrencetime}
\end{equation}
where $S$ is the thermodynamic entropy.
\end{consequence}

\begin{proof}
By the Poincar\'e Recurrence Consequence (\cref{ch:oscillation}), the
trajectory of a bounded system returns to any neighbourhood of the initial
point infinitely often.  Equilibrium is attained when the trajectory has
traversed its accessible phase space sufficiently to establish
time-averaged properties identical to phase-space averages (Birkhoff's
ergodic theorem).  The recurrence time is bounded below by the volume of
accessible phase space divided by the smallest distinguishable
neighbourhood:
\[
  \tau_P \sim \exp(S/\kB)
\]
by the Boltzmann relation $S = \kB \ln W$ with $W$ the number of
accessible microstates.
\end{proof}

\begin{consequence}{Counting irreversibility}{countingirrev}
The probability that a spontaneous macroscopic reversal of entropy
increase occurs in a bounded system with $N_{\mathrm{state}}$ accessible
categorical states is
\begin{equation}
  P_{\mathrm{rev}} \sim \exp(-N_{\mathrm{state}}).
  \label{eq:irrev}
\end{equation}
\end{consequence}

\begin{proof}
A macroscopic reversal requires that the categorical-completion trajectory
(time emergence, \cref{ch:oscillation}) traverse all $N_{\mathrm{state}}$
completions in reverse order.  By the monotonicity of completion, this
requires each completion to be ``un-completed'', which under
measure-preserving dynamics has probability equal to the measure of the
pre-completion cell within the total phase space.  The joint probability
over $N_{\mathrm{state}}$ independent completions is
$\exp(-N_{\mathrm{state}})$.
\end{proof}

\begin{remark}[Arrow of time as a structural counting statement]

statistical statement to a structural counting statement: the arrow of

\end{remark}

\section*{Appendix to Part III: Thermodynamic Consistency Checks}

We verify that the categorical thermodynamic variables defined in Part III
are consistent with the four classical thermodynamic identities: the
Clausius inequality, the Gibbs free energy, the fluctuation--dissipation
theorem, and Onsager reciprocity.

%------------------------------------------------------------
\subsection{Consistency with Clausius}
%------------------------------------------------------------

The Clausius inequality $\oint \mathrm{d}S \ge \oint \mathrm{d}Q/T$ follows
from the monotonicity of categorical entropy (from \cref{ch:sentropy})
combined with the First Law $\mathrm{d}U = \mathrm{d}Q - P\mathrm{d}V$.

For a \emph{reversible} cycle, $\mathrm{d}S_{\mathrm{cat}} = \mathrm{d}Q/T$
at each step, so the integral around the cycle is zero.  For an
\emph{irreversible} cycle, at least one step increases categorical entropy
without a corresponding heat transfer, giving strict inequality.

Both outcomes are special cases of the Axiom applied through the categorical
monotonicity: $\mathrm{d}S_{\mathrm{cat}} \ge 0$ (Corollary~\ref{cor:twosecondlaws}),
which at each step requires
\[
  \mathrm{d}S_{\mathrm{cat}} \ge \frac{\mathrm{d}Q}{T}.
\]
This is the Clausius inequality, derived from the partition structure
without additional postulate.

%------------------------------------------------------------
\subsection{Consistency with Gibbs}
%------------------------------------------------------------

The Gibbs free energy $G = H - TS$ is minimised at thermodynamic
equilibrium at constant $T$ and pressure.  In the categorical framework:
\begin{itemize}[nosep]
  \item $H$ (enthalpy) is the total categorical energy including the $PV$
    term from Consequence~\ref{cons:idealgasN} (\cref{ch:idealgas});
  \item $S$ is the categorical entropy $\St$ from
    Consequence~\ref{cons:tripleent-thermo} (\cref{ch:idealgas});
  \item $T$ is the categorical temperature (Definition~\ref{def:cattemp}
    of \cref{ch:idealgas}).
\end{itemize}
Minimising $G$ at fixed $T$ and $P$ gives the equilibrium distribution of
partition-cell occupations, which coincides with the Boltzmann
distribution in the high-temperature limit, and with Fermi--Dirac or
Bose--Einstein statistics in the quantum limits (covered by
Consequence~\ref{cons:universaleos} above).

\subsection{Consistency with the fluctuation--dissipation theorem}
%------------------------------------------------------------

The fluctuation--dissipation theorem (FDT) relates the linear response
function to thermal fluctuation correlations:
\[
  \chi''(\omega)
  = \frac{\omega}{2\kB T}\,S(\omega),
\]
where $\chi''$ is the imaginary part of the response function and $S(\omega)$
is the spectral density of fluctuations.

In the partition framework:
\begin{itemize}[nosep]
  \item The linear response is the transport sum~\eqref{eq:transport},
    expressed as a function of frequency via Laplace transform of the
    partition-lag time distribution.
  \item The thermal fluctuations are the categorical transitions averaged
    over equilibrium histories, whose spectral density is determined by
    the same partition-lag times $\taup^{(ij)}$.
\end{itemize}
The FDT then becomes a statistical relationship among the partition-lag
times $\taup^{(ij)}$ that enter both $\chi''(\omega)$ and $S(\omega)$.
The standard derivation (Kubo, 1957) proceeds unchanged with
partition-lag times as the relaxation parameters.

\subsection{Consistency with Onsager reciprocity}
%------------------------------------------------------------

Onsager reciprocity states that the transport coefficient matrix
$L_{ij}$ defined by $J_i = \sum_j L_{ij} X_j$ (thermodynamic forces
$X_j$ driving fluxes $J_i$) is symmetric: $L_{ij} = L_{ji}$.

In the categorical framework, the matrix elements are
\[
  L_{ij}
  = \frac{1}{\mathcal{N}}\sum_{k,l}\taup^{(kl)}\,g_{ij}^{(kl)},
\]
where $g_{ij}^{(kl)}$ is the coupling between force $j$ and flux $i$
at transition $k \to l$.  Time-reversal symmetry of the Hamiltonian flow
(guaranteed by the Axiom, \cref{ch:axiom}, since the bounded phase space
admits a symplectic structure) requires $\taup^{(kl)} = \taup^{(lk)}$
and $g_{ij}^{(kl)} = g_{ji}^{(kl)}$, giving $L_{ij} = L_{ji}$.
This is Onsager reciprocity, derived from the time-reversal symmetry of
the partition dynamics.

\subsection{Stokes--Einstein relation}

\begin{consequence}{Stokes--Einstein relation}{stokeseinstein}
For a particle of radius $r$ in a fluid of viscosity $\eta$, the diffusion coefficient is
\begin{equation}
D=\frac{\kB T}{6\pi\eta r}.
\label{eq:stokeseinstein}
\end{equation}
\end{consequence}

\begin{proof}
By Consequence~\ref{cons:transport} in the diffusivity case, $D=\kB T\taup/m$. The partition-lag time $\taup$ for a sphere in a viscous fluid is set by the Stokes drag coefficient $\zeta=6\pi\eta r$: $\taup=m/\zeta$. Substituting, $D=\kB T/(6\pi\eta r)$, which is~\eqref{eq:stokeseinstein}.
\end{proof}

\subsection{Bose--Einstein condensation}

A Bose gas at temperature below the critical temperature
\begin{equation*}
T_{c}=\frac{1}{\kB}\left(\frac{n}{\zeta(3/2)}\right)^{2/3}\cdot\frac{2\pi\hbar^{2}}{m}
\end{equation*}
undergoes Bose--Einstein condensation \citep{bose1924,andersonetal1995}: a macroscopic fraction of the particles occupies the single lowest partition cell. By Consequence~\ref{cons:partitionextinction}, this extinction of partition distinguishability leads to zero viscosity (superfluidity) for the condensate. Observations of BEC in dilute atomic gases \citep{andersonetal1995} confirm this prediction.

\subsection{Superconductivity}

\begin{consequence}{Superconducting gap}{scgap}
In a superconductor at $T<T_{c}$, the excitation gap $\Delta(T)$ and the resistivity $\rho(T)$ obey
\begin{equation*}
\rho(T<T_{c})=0 \text{ exactly.}
\end{equation*}
\end{consequence}

\begin{proof}
At $T<T_{c}$, the Cooper-pair partition coalesces into a single cell (Consequence~\ref{cons:partitionextinction}); the transport sum in Consequence~\ref{cons:transport} has no active terms. $\rho=0$ exactly.
\end{proof}

Observations \citep{onnes1911} confirm $\rho=0$ (to the precision of persistent-current measurements, $\rho<10^{-23}$ $\Omega$m) below $T_{c}$ in mercury, lead, niobium, and many other elemental superconductors.
