\section{Categorical Temperature, Pressure, and the Ideal Gas Law}
\label{sec:ch16-idealgas}

Define thermodynamic quantities from partition coordinates:
\begin{align}
  T_\mathrm{cat} &= \frac{\partial E}{\partial \Sk}\bigg|_V
                 = \frac{\hbar\omega_k}{\kB} && \text{(categorical temperature)}, \\
  P_\mathrm{cat} &= -\frac{\partial E}{\partial V}\bigg|_S
                 = \frac{E}{V}           && \text{(categorical pressure)}.
\end{align}

\begin{consequence}{Ideal Gas Law for $N \geq 1$}{idealgasN}
  For any $N \geq 1$ non-interacting bounded oscillators:
  \begin{equation}
    P_\mathrm{cat} V = N\kB T_\mathrm{cat}.
  \end{equation}
\end{consequence}

\begin{proof}[Proof sketch]
  Sum the partition energies $E_k = \hbar\omega_k$ over the discrete Koopman
  spectrum; apply the equipartition theorem to the categorical temperature
  definition.
  The ideal gas law emerges without the assumption of point particles,
  three-dimensional space, or any particular equation of state.
  The derivation holds for $N = 1$ (a single bounded oscillator).
\end{proof}

The ideal gas law holds even for $N = 1$ because it is a statement about the
partition structure, not about the thermodynamic limit $N \to \infty$.

\section{Bounded Maxwell--Boltzmann Distribution}
\label{sec:ch16-mb}

\begin{consequence}{Bounded Maxwell--Boltzmann}{MBbounded}
  The velocity distribution of particles in a bounded phase space with
  propagation bound $c$ is
  \begin{equation}
    f(v) \propto \exp\!\left(-\frac{m_0 c^2}{2\kB T}
      \cdot \frac{v^2/c^2}{1 - v^2/c^2}\right),
  \end{equation}
  which reduces to the Maxwell--Boltzmann distribution $f(v) \propto
  e^{-mv^2/2\kB T}$ for $v \ll c$.
\end{consequence}

The relativistic correction is an exact consequence of the propagation bound
$c$ (Consequence~\ref{cons:propagationbound}); the non-relativistic limit
recovers the standard Maxwell--Boltzmann distribution.
